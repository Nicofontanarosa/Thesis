# 1. PORT-BASED PROTOCOLS
# Format: tcp:<port>,udp:<port>@<ProtoName>[=ProtoID][,cat=N][,breed=M]
#tcp:80@HTTP=1,cat=1,breed=1
#tcp:443@TLS=2,cat=2,breed=1
#udp:5060@SIP=3,cat=3,breed=1
#tcp:65535@TestProto=4000,cat=9,breed=1

# 2. HOST-BASED (hostname/SNI)
# Format: host:"<value>"@<ProtoName>[=ProtoID][,cat=N][,breed=M]
#host:"disneyplus.com",host:"*.disneyplus.net"@DisneyPlus=4200,cat=11,breed=1
#host:"api-global.netflix.com"@Netflix=4201,cat=11,breed=2

# 3. IP-BASED (IPv4)
# Format: ip:<addr>[:<port>]@<ProtoName>[=ProtoID][,cat=N][,breed=M]
#ip:8.248.73.247:443@AmazonPrime=4300,cat=12,breed=1
#ip:3.3.3.3:443@CustomProtocolA=4301,cat=5,breed=2

# 4. IPV6-BASED
# Format: ipv6:[<addr>][:<port>]@<ProtoName>[=ProtoID][,cat=N][,breed=M]
#ipv6:[2001:db8::c2fd:b817:5ca8:82dd]:16690@CustomProtocolJ=4400,cat=6,breed=3
#ipv6:[fe80::76ac:b9ff:fe6c:c124]:12717@CustomProtocolG=4401,cat=6,breed=4

# 5. SYMBOLIC IP / HOSTNAME
# Format: ip:<hostname>@<ProtoName>[=ProtoID][,cat=N][,breed=M]
#ip:www.ntop.org@ntop=4500,cat=7,breed=1
#ipv6:www.ntop.org@ntop=4501,cat=7,breed=1

# 6. JA4 FINGERPRINTS
# Format: ja4:<fingerprint>@<ProtoName>[=ProtoID][,cat=N][,breed=M]
#ja4:t12d1409h1_c866b44c5a26_b39be8c56777@CustomJA4Proto=2060,cat=2,breed=5

# 7. CERTIFICATION
# Custom certification autorities we trust
#trusted_issuer_dn:"CN=813845657003339838, O=Code42, OU=TEST, ST=MN, C=US"@CustomProto=2060,cat=2,breed=5

# 8. nDPI FINGERPRINTS
# Format: ndpifp:<fingerprint>@<ProtoName>[=ProtoID][,cat=N][,breed=M]
#ndpifp:8ae775daeb30f7aa5790706158288639@CustomnDPIFPProto=2061,cat=3,breed=2
#ndpifp:2d3ea0b7f090060fa1b9f3783362f862@CustomnDPIFPProto2=2062,cat=3,breed=3

# NDPI_PROTOCOL_ACCEPTABLE = 1,             /* Probably doesn't provide risks, but could be malicious (e.g., Dropbox) */
# NDPI_PROTOCOL_FUN = 2,                    /* Pure fun protocol, which may be prohibited by the user policy (e.g., Netflix) */
# NDPI_PROTOCOL_UNSAFE = 3,                 /* Probably provides risks, but could be a normal traffic. Unencrypted protocols with clear pass should be here (e.g., telnet) */
