def BFS_search_flows(flows, word):

    results = []
    visited = set()

    # BFS
    to_visit = []

    # phase 1: find initial matches
    for i, flow in enumerate(flows):
        if flow.get("proto_field", "").upper() == "5/DNS":
            continue
        for v in flow.values():
            if word.lower() in str(v).lower():
                to_visit.append(i)

    # phase 2: BFS on related attributes
    while to_visit:
        idx = to_visit.pop(0)
        if idx in visited:
            continue
        visited.add(idx)
        flow = flows[idx]
        if flow.get("proto_field", "").upper() == "5/DNS":
            continue
        results.append(flow)

        related = [flow.get(key) for key in config.CLUSTER_KEY]

        # find other flows with related attributes
        for j, other in enumerate(flows):
            if j in visited:
                continue
            for key in related:
                if key and key != "N/A" and key in other.values():
                    to_visit.append(j)

    return results

# key to use for clustering flows
CLUSTER_KEY = ["ip_destination", "sni"]
